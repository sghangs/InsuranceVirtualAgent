CREATE EXTENSION IF NOT EXISTS vector;

-- Conversations table: long-term memory and chat history (uses pgvector for semantic search)
CREATE TABLE conversations (
    conversation_id UUID PRIMARY KEY DEFAULT gen_random_uuid(), -- unique conversation/message ID
    user_id UUID REFERENCES users(user_id),                      -- links to user
    thread_id UUID,                                             -- for mapping to chat session (short-term memory)
    policy_number VARCHAR(50),                                  -- optional, for insurance context
    message TEXT NOT NULL,                                      -- the actual message or summary
    embedding VECTOR(1536),                                     -- pgvector column for semantic search (adjust dimension as needed)
    created_at TIMESTAMP DEFAULT NOW()
);

**** Create index for conversations table 
CREATE INDEX idx_embedding ON conversations USING ivfflat (embedding vector_cosine_ops);

-- User Profiles table: persistent user info for insurance agent
CREATE TABLE user_profiles (
    user_id UUID PRIMARY KEY DEFAULT gen_random_uuid(), -- persistent unique user ID
    name VARCHAR(200),
    email VARCHAR(200),
    phone VARCHAR(20),
    dob DATE,
    policy_number VARCHAR(50),
    policy_type VARCHAR(100),
    address TEXT,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

-- Enable UUID extension:
CREATE EXTENSION IF NOT EXISTS "pgcrypto";

CREATE TABLE users (
    user_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(200),
    email VARCHAR(200) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    created_at TIMESTAMP DEFAULT NOW()
);
